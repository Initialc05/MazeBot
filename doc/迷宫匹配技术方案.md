# 迷宫匹配技术方案

## 1. 项目概述

### 1.1 目标
实现基于雷达数据的迷宫墙壁检测和拟合，建立4×4网格迷宫模型，为后续的自动寻路算法提供准确的环境信息。

### 1.2 核心功能
- **墙壁检测**：通过雷达点云数据检测迷宫墙壁
- **线段拟合**：使用数学方法拟合墙壁线段
- **置信度评估**：为每个检测结果提供可靠性评分
- **网格建模**：将检测结果映射到4×4网格结构
- **实时可视化**：提供直观的地图显示和调试界面

#### 3.2.1 数据预处理
```python
def preprocess_lidar_data(lidar_data):
    """
    雷达数据预处理
    
    输入：[(angle, distance, quality), ...]
    输出：过滤后的高质量点云数据
    """
    filtered_data = []
    for angle, distance, quality in lidar_data:
        # 质量过滤：quality > 5
        # 距离过滤：0.1m < distance < 1.5m
        if quality > 5 and 0.1 < distance < 1.5:
            filtered_data.append((angle, distance, quality))
    return filtered_data
```

#### 3.2.2 坐标转换
```python
def polar_to_cartesian(angle, distance, robot_pos, robot_heading):
    """
    极坐标转笛卡尔坐标
    
    参数：
    - angle: 雷达角度
    - distance: 距离
    - robot_pos: 机器人位置 (x, y)
    - robot_heading: 机器人朝向角度
    
    返回：(world_x, world_y)
    """
    absolute_angle = angle + robot_heading
    world_x = robot_pos[0] + distance * cos(radians(absolute_angle))
    world_y = robot_pos[1] + distance * sin(radians(absolute_angle))
    return world_x, world_y
```

#### 3.2.3 方向聚类
```python
def cluster_points_by_direction(world_points, robot_pos):
    """
    按方向聚类点云
    
    将点云按四个主要方向分组：
    - 北方向：0° ± 22.5°
    - 东方向：90° ± 22.5°
    - 南方向：180° ± 22.5°
    - 西方向：270° ± 22.5°
    """
    clusters = {'north': [], 'east': [], 'south': [], 'west': []}
    
    for x, y, quality in world_points:
        # 计算相对于机器人的角度
        dx = x - robot_pos[0]
        dy = y - robot_pos[1]
        angle = degrees(atan2(dy, dx))
        angle = (angle + 360) % 360  # 归一化到[0, 360)
        
        # 分配到对应方向
        if 337.5 <= angle or angle < 22.5:
            clusters['north'].append((x, y, quality))
        elif 67.5 <= angle < 112.5:
            clusters['east'].append((x, y, quality))
        elif 157.5 <= angle < 202.5:
            clusters['south'].append((x, y, quality))
        elif 247.5 <= angle < 292.5:
            clusters['west'].append((x, y, quality))
    
    return clusters
```

### 3.3 线段拟合算法

#### 3.3.1 最小二乘法拟合
```python
def fit_line_segment(points, direction):
    """
    使用最小二乘法拟合直线段
    
    参数：
    - points: 点云数据 [(x, y, quality), ...]
    - direction: 方向 ('north', 'east', 'south', 'west')
    
    返回：WallSegment对象
    """
    if len(points) < 3:
        return None
    
    x_coords = [p[0] for p in points]
    y_coords = [p[1] for p in points]
    qualities = [p[2] for p in points]
    
    if direction in ['north', 'south']:
        # 水平墙壁：拟合 y = constant
        y_mean = average(y_coords, weights=qualities)
        confidence = calculate_confidence(points, 'horizontal', y_mean)
        
        x_min, x_max = min(x_coords), max(x_coords)
        start_pos = (x_min, y_mean)
        end_pos = (x_max, y_mean)
    else:  # east, west
        # 垂直墙壁：拟合 x = constant
        x_mean = average(x_coords, weights=qualities)
        confidence = calculate_confidence(points, 'vertical', x_mean)
        
        y_min, y_max = min(y_coords), max(y_coords)
        start_pos = (x_mean, y_min)
        end_pos = (x_mean, y_max)
    
    return WallSegment(start_pos, end_pos, confidence)
```

#### 3.3.2 置信度计算
```python
def calculate_confidence(points, wall_type, fitted_value):
    """
    计算拟合置信度
    
    置信度 = (点数贡献 + 拟合质量贡献) / 2
    
    点数贡献 = min(1.0, 点数 / 10.0)
    拟合质量贡献 = max(0.0, 1.0 - 标准差 / 0.1)
    """
    if len(points) < 3:
        return 0.0
    
    # 计算点到拟合线的距离
    distances = []
    for x, y, quality in points:
        if wall_type == 'horizontal':
            distance = abs(y - fitted_value)
        else:  # vertical
            distance = abs(x - fitted_value)
        distances.append(distance)
    
    # 计算标准差
    std_distance = std(distances)
    
    # 置信度计算
    point_confidence = min(1.0, len(points) / 10.0)
    quality_confidence = max(0.0, 1.0 - std_distance / 0.1)
    
    return (point_confidence + quality_confidence) / 2.0
```

### 3.4 网格映射算法

#### 3.4.1 墙壁映射
```python
def map_walls_to_grid(fitted_walls, maze_map):
    """
    将拟合的墙壁线段映射到网格墙壁
    
    参数：
    - fitted_walls: 拟合的墙壁线段列表
    - maze_map: 迷宫地图对象
    """
    for wall_segment in fitted_walls:
        if abs(wall_segment.start_pos[1] - wall_segment.end_pos[1]) < 0.1:
            # 水平墙壁
            y = wall_segment.start_pos[1]
            x_start = wall_segment.start_pos[0]
            x_end = wall_segment.end_pos[0]
            
            # 映射到网格
            grid_y = round(y)
            if 0 <= grid_y < 5:
                for x in arange(x_start, x_end, 0.1):
                    grid_x = round(x - 0.5)
                    if 0 <= grid_x < 4:
                        if (maze_map.walls_h[grid_x][grid_y] is None or 
                            maze_map.walls_h[grid_x][grid_y].confidence < wall_segment.confidence):
                            maze_map.walls_h[grid_x][grid_y] = wall_segment
        else:
            # 垂直墙壁
            x = wall_segment.start_pos[0]
            y_start = wall_segment.start_pos[1]
            y_end = wall_segment.end_pos[1]
            
            # 映射到网格
            grid_x = round(x)
            if 0 <= grid_x < 5:
                for y in arange(y_start, y_end, 0.1):
                    grid_y = round(y - 0.5)
                    if 0 <= grid_y < 4:
                        if (maze_map.walls_v[grid_x][grid_y] is None or 
                            maze_map.walls_v[grid_x][grid_y].confidence < wall_segment.confidence):
                            maze_map.walls_v[grid_x][grid_y] = wall_segment
```

#### 6.3.1 队列设计
```python
class MazeInterface:
    def __init__(self):
        # 数据队列
        self.lidar_queue = Queue()
        self.odom_queue = Queue()
        
        # 更新线程
        self.update_thread = None
        self.is_running = False
    
    def _update_loop(self):
        """更新循环"""
        while self.is_running:
            # 处理雷达数据队列
            lidar_batch = []
            while not self.lidar_queue.empty():
                lidar_data = self.lidar_queue.get_nowait()
                lidar_batch.extend(lidar_data)
            
            # 处理里程计数据队列
            robot_pos = None
            robot_heading = None
            while not self.odom_queue.empty():
                odom_data = self.odom_queue.get_nowait()
                robot_pos = (odom_data['x'], odom_data['y'])
                robot_heading = odom_data['heading']
            
            # 更新迷宫地图
            if lidar_batch and robot_pos is not None:
                self.maze_mapper.update_maze_map(lidar_batch, robot_pos, robot_heading)
```
